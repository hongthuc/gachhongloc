        /* Slider 2 Banner tự trượt */
        let currentSlide = 0;
        const slides = document.querySelectorAll('.banner-slide');
        let slideInterval = null;

        function showSlide(index) {
            currentSlide = (index + slides.length) % slides.length;
            slides.forEach((s, i) => {
                if (i === currentSlide) {
                    s.classList.remove('opacity-0', 'z-0');
                    s.classList.add('opacity-100', 'z-10');
                } else {
                    s.classList.remove('opacity-100', 'z-10');
                    s.classList.add('opacity-0', 'z-0');
                }
            });
            const d0 = document.getElementById('dot-0');
            const d1 = document.getElementById('dot-1');
            if (currentSlide === 0) {
                d0.className = 'w-6 h-2 rounded-full bg-amber-400 transition-all';
                d1.className = 'w-2.5 h-2 rounded-full bg-white/50 transition-all';
            } else {
                d0.className = 'w-2.5 h-2 rounded-full bg-white/50 transition-all';
                d1.className = 'w-6 h-2 rounded-full bg-amber-400 transition-all';
            }
        }

        function changeSlide(step) {
            showSlide(currentSlide + step);
            resetSlideTimer();
        }

        function goToSlide(index) {
            showSlide(index);
            resetSlideTimer();
        }

        function resetSlideTimer() {
            if (slideInterval) clearInterval(slideInterval);
            slideInterval = setInterval(() => {
                showSlide(currentSlide + 1);
            }, 4500);
        }
